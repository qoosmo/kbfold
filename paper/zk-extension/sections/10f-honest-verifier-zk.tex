\subsection{A concrete perfect honest-verifier simulator}\label{sec:realize:hvzk}

\begin{definition}[Restricted honest-sigma-verifier model]\label{def:realize:hv-model}
Consider a single opening of the Chapter~9 public-core
protocol with the bit commitments of
Definition~\ref{def:realize:field-commit}.
Every commitment uses fresh independent uniform
Pedersen blinding, and every prescribed bitness
or affine-parity proof uses fresh independent
Schnorr OR-prover randomness.
Each \emph{sigma-proof challenge} is an
independent uniform element of $\mathbb Z_P$,
sampled after the corresponding OR first message.
The outer Ring Switching challenges $\theta,z$
may still be arbitrary causal functions of the
entire visible history, including commitments and
earlier sigma transcripts.
The verifier is not permitted to request
extraneous secret-dependent predicates or
actual commitment openings.
This is the \emph{honest-sigma-verifier model}.
\end{definition}

\begin{lemma}[Fresh commitments are conditionally uniform]\label{lem:realize:conditional-commit}
At every registration stage, conditional on the
previous visible history and on any actual
private bit vector chosen at that stage,
the fresh Pedersen commitments are jointly
uniform on the corresponding power of $G$.
This conditional law is independent of the
committed bit values.
\end{lemma}
\begin{proof}
The fresh commitment blindings are chosen
independently uniformly from $\mathbb Z_P$
after the prior history. Once the bits
are fixed, Lemma~\ref{lem:realize:hiding}
makes every commitment uniform on $G$
and the full vector independently uniform.
Because the law is the same for every bit
vector, averaging over any distribution
of private bits conditional on the history
does not change the uniform law.
\end{proof}

\begin{lemma}[A prescribed OR proof has a witness-free conditional simulator]\label{lem:realize:conditional-or}
For a public OR-proof statement determined by
the current visible history and any fixed
valid hidden witness, if the new global sigma
challenge is independent uniform, then the
conditional transcript law equals that of
Theorem~\ref{thm:realize:or-hvzk},
regardless of the witness and preceding
proof transcripts.
\end{lemma}
\begin{proof}
All OR-protocol randomizers for the current
proof are fresh and independent of the
previous history. The prover's witness
may be a deterministic function of
previously registered openings and current
public coefficients, but the proof
of Theorem~\ref{thm:realize:or-hvzk}
is pointwise in every fixed witness and
every fixed public statement.
Thus the same perfect simulation equality
holds after conditioning on the current
history and witness. Since its conditional
law depends only on the statement,
averaging over the witness preserves it.
\end{proof}

\begin{definition}[Concrete honest-verifier simulator]\label{def:realize:hv-sim}
The simulator operates at the same causal protocol
stages as the real implementation:
\begin{enumerate}[label=\textup{(\roman*)}]
\item at each hidden-bit registration, it samples
independent uniform elements of $G$ as the
public commitments without choosing openings;
\item at each prescribed bitness or affine-parity
OR proof, it uses the \emph{witness-free}
transcript simulator of
Theorem~\ref{thm:realize:or-hvzk}
on the publicly known commitments and candidates;
\item at public-core stages it uses the
causal simulator of
Definition~\ref{def:zkrs:core-sim};
\item it forwards the verifier's allowed
adaptive outer challenges and returns only
the honest constant acceptance bits.
\end{enumerate}
The simulator is defined for the honest-sigma-verifier
model of Definition~\ref{def:realize:hv-model}.
\end{definition}

\begin{theorem}[Perfect honest-verifier ZK of the concrete affine transcript]\label{thm:realize:full-hvzk}
For one admissible Ring Switching opening and
every causal verifier obeying
Definition~\ref{def:realize:hv-model},
the joint visible transcript of all
fresh Pedersen bit commitments,
prescribed Schnorr OR bitness and
affine-parity proofs, and the public
core $(m,y',\theta,U,z,A)$ has exactly
the same law as the transcript generated by
Definition~\ref{def:realize:hv-sim}.
The equality holds for any two admissible
witnesses with the same public instance.
It excludes an external polynomial-commitment
scheme and its authenticated openings.
\end{theorem}
\begin{proof}
Include the verifier's random tape in the
finite visible state and induct over the
ordered message stages.
For the initial hidden-variable
registrations, Lemma~\ref{lem:realize:conditional-commit}
makes each fresh commitment vector
conditionally uniform independent of
the hidden bit values, exactly as in
the simulator.
For each prescribed OR proof,
Lemma~\ref{lem:realize:conditional-or}
identifies the complete conditional
transcript law with the simulator's
witness-free OR-proof law, even when
the public statement depends on
earlier messages or outer challenges.
For the intervening public core,
Theorem~\ref{thm:zkrs:perfect-core}
gives the witness-free conditional
distributions for $(m,y')$, $U$,
and $A$. Conditioning on previous
fresh perfectly hiding commitments and
their perfectly honest-verifier-simulated
proof transcripts cannot disclose
the underlying private masks: the
conditional laws established in the
two preceding lemmas do not depend on
those masks. Therefore the three
public-core transition laws remain
unchanged under that additional
conditioning.
At stages where the verifier selects
an outer challenge, the same causal
verifier transition kernel is applied
to identically distributed visible
histories in both experiments.
Every prescribed honest affine
statement is valid by
Lemmas~\ref{lem:zkrs:honest-checker}
and~\ref{lem:zkrs:honest-relations},
and the concrete proofs are perfectly
complete by
Theorem~\ref{thm:realize:compiler-complete};
hence the associated acceptance
bits are always $1$.
Thus every transition in the
real and simulated finite transcript
processes has the same conditional
law. Induction on their length
establishes exact equality of the
complete visible joint laws.
\end{proof}

\begin{corollary}[No witness-dependent commitment-token distribution]\label{cor:realize:tokens}
Under the hypotheses of
Theorem~\ref{thm:realize:full-hvzk},
the complete vector of registered
Pedersen commitment tokens is jointly
independent of the hidden witness.
\end{corollary}
\begin{proof}
It is a deterministic projection
of the perfectly simulated visible
transcript of that theorem.
Alternatively, iterate
Lemma~\ref{lem:realize:conditional-commit}
over all registration stages.
\end{proof}

\begin{remark}[Exactly what has been realized]
The theorem gives \emph{perfect honest-verifier
zero knowledge} for the explicit
bit-commitment and affine-proof transcripts,
with adaptive \emph{outer} challenges.
It is not full malicious-verifier ZK,
because the sigma-proof verifier
challenges are assumed independently
uniform. It is not an end-to-end KBFold
PCS theorem, because the original
folded oracle commitment, Merkle
authentication paths, and masked
sumcheck transcript are absent.
The commitment is classical
discrete-logarithm based and is
not post-quantum binding.
\end{remark}
